# Dark-mode UI overhaul — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the inline-styled, browser-default UI with a coherent dark theme (CSS-variable design tokens, reusable `<Button>` + `<Icon>` components, Wizard stepper) and bundle a discovered keychain bug fix so the OCR flow actually works.

**Architecture:** Pure presentational refactor. One CSS file holds tokens. Two new presentational components (`Button`, `Icon`) plus one composite (`Stepper`). Pages swap raw `<button>`/`<Link>` for these components. No business-logic changes. One Rust dep flag fix (keyring → enable platform-native backends) is included as Task 0 because the OCR flow is unusable without it.

**Tech Stack:** React 18 + TypeScript + Vite + Tauri 2 (frontend); Rust + keyring crate (backend). Inline SVG icons, plain CSS. No new npm or cargo crates beyond enabling keyring features that are already in the published crate.

**Spec:** `docs/superpowers/specs/2026-05-19-dark-mode-ui-design.md`

```tsx
import "./Button.css";
import type { ButtonHTMLAttributes, ReactNode } from "react";

type Variant = "primary" | "secondary" | "ghost" | "danger";
type Size = "md" | "sm";

interface ButtonProps extends ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: Variant;
  size?: Size;
  leftIcon?: ReactNode;
  rightIcon?: ReactNode;
  iconOnly?: boolean;
}

export function Button({
  variant = "secondary",
  size = "md",
  leftIcon,
  rightIcon,
  iconOnly = false,
  className,
  children,
  ...rest
}: ButtonProps) {
  const classes = [
    "btn",
    `btn--${variant}`,
    size === "sm" ? "btn--sm" : "",
    iconOnly ? "btn--iconOnly" : "",
    className ?? "",
  ].filter(Boolean).join(" ");

  return (
    <button className={classes} {...rest}>
      {leftIcon}
      {!iconOnly && children}
      {rightIcon}
    </button>
  );
}
```

```tsx
import { Fragment } from "react";
import "./Stepper.css";

interface Props {
  steps: string[];
  current: number; // 1-based
}

export function Stepper({ steps, current }: Props) {
  return (
    <nav className="stepper" aria-label="Progress">
      {steps.map((label, i) => {
        const stepNumber = i + 1;
        const state =
          stepNumber < current ? "done" :
          stepNumber === current ? "current" : "upcoming";
        return (
          <Fragment key={label}>
            <div
              className={`stepper__item stepper__item--${state}`}
              aria-current={state === "current" ? "step" : undefined}
            >
              <div className="stepper__dot">{stepNumber}</div>
              <div className="stepper__label">{label}</div>
            </div>
            {i < steps.length - 1 && (
              <div
                className={`stepper__connector ${stepNumber < current ? "stepper__connector--done" : ""}`}
              />
            )}
          </Fragment>
        );
      })}
    </nav>
  );
}
```

```tsx
import { useEffect, useState } from "react";
import { useNavigate } from "react-router-dom";
import { api } from "../lib/tauri";
import { Button } from "../components/Button";
import { Icon } from "../components/Icon";

export default function Settings() {
  const navigate = useNavigate();
  const [key, setKey] = useState("");
  const [hasKey, setHasKey] = useState<boolean | null>(null);
  const [saved, setSaved] = useState(false);
  const [err, setErr] = useState<string | null>(null);

  useEffect(() => {
    api.getApiKey().then((k) => setHasKey(!!k));
  }, []);

  async function save() {
    setErr(null);
    try {
      await api.setApiKey(key);
      setSaved(true);
      setHasKey(true);
      setKey("");
    } catch (e: any) {
      setErr(String(e?.message ?? e));
    }
  }

  async function remove() {
    try {
      await api.deleteApiKey();
      setHasKey(false);
    } catch (e: any) {
      setErr(String(e?.message ?? e));
    }
  }

  async function backOrSaveAndBack() {
    if (key.length > 0) {
      await save();
      if (err) return; // save failed — stay on page
    }
    navigate("/");
  }

  const dirty = key.length > 0;

  return (
    <div style={{ padding: 32, maxWidth: 600, margin: "0 auto" }}>
      <Button
        variant={dirty ? "primary" : "ghost"}
        leftIcon={<Icon name={dirty ? "check" : "arrow-left"} />}
        onClick={backOrSaveAndBack}
      >
        {dirty ? "Save & Back" : "Back"}
      </Button>

      <h1 style={{ marginTop: 24 }}>Settings</h1>

      <h3>Anthropic API key</h3>
      <p style={{ color: "var(--text-muted)" }}>
        Stored in your OS keychain. Used for receipt OCR via Claude.
      </p>
      <p style={{ display: "inline-flex", alignItems: "center", gap: 6 }}>
        {hasKey === null ? (
          <span style={{ color: "var(--text-muted)" }}>Checking…</span>
        ) : hasKey ? (
          <>
            <Icon name="check" />
            <span style={{ color: "var(--success)" }}>Key configured</span>
          </>
        ) : (
          <>
            <Icon name="x" />
            <span style={{ color: "var(--text-muted)" }}>No key set</span>
          </>
        )}
      </p>

      <input
        type="password"
```
